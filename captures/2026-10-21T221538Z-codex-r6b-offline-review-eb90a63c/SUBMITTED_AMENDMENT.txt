from pathlib import Path
import hashlib
import json
import numpy as np

def verify_manifest_and_load_payloads(index_path: Path):
    """
    Corrects Defect 2: Parses upstream[].files and loads .npz containers.
    """
    with open(index_path, "r", encoding="utf-8") as f:
        index_data = json.load(f)

    verified_files = 0
    npz_payloads = {}

    # Traverse upstream[].files schema
    upstream_list = index_data.get("upstream", [])
    for entry in upstream_list:
        files = entry.get("files", [])
        for file_info in files:
            file_rel_path = file_info["path"]
            expected_sha256 = file_info["sha256"]
            file_path = index_path.parent / file_rel_path

            if not file_path.exists():
                raise FileNotFoundError(f"Missing pinned file: {file_path}")

            # Verify byte-exact SHA-256
            hasher = hashlib.sha256()
            with open(file_path, "rb") as bf:
                while chunk := bf.read(65536):
                    hasher.update(chunk)
            computed_sha256 = hasher.hexdigest()

            if computed_sha256 != expected_sha256:
                raise ValueError(
                    f"Hash mismatch on {file_rel_path}: "
                    f"computed {computed_sha256} != expected {expected_sha256}"
                )
            verified_files += 1

            # Ingest .npz archives
            if file_path.suffix == ".npz":
                with np.load(file_path) as archive:
                    npz_payloads[file_rel_path] = {k: archive[k] for k in archive.files}

    return verified_files, npz_payloads

def evaluate_mode_verification(
    A_full: np.ndarray,
    v: np.ndarray,
    lambda_solver: float,
    gamma_module: float,
    tol: float = 5e-12
):
    """
    Corrects Defects 1 & 3:
    - Direct residual vector norm ||r||_2 prevents catastrophic cancellation.
    - Frozen acceptance criterion evaluated against A_full and lambda_solver.
    - Rayleigh quotient and Temple enclosure evaluated separately for variational proof.
    """
    # Normalize vector
    v_norm = v / np.linalg.norm(v)

    # 1. FROZEN ACCEPTANCE RULE (PR352/PR358 contract)
    # Check: ||A_full @ v - lambda_solver * v||_2 / max(1, ||A_full||_inf + |lambda_solver|) <= tol
    A_full_inf = float(np.max(np.sum(np.abs(A_full), axis=1)))
    scale_factor = max(1.0, A_full_inf + abs(lambda_solver))

    r_solver = (A_full @ v_norm) - (lambda_solver * v_norm)
    residual_solver_norm = float(np.linalg.norm(r_solver))
    scaled_solver_residual = residual_solver_norm / scale_factor
    acceptance_passed = bool(scaled_solver_residual <= tol)

    # 2. VARIATIONAL ANALYSIS & TEMPLE ENCLOSURE (Numerical Stability Fix)

    theta = float(np.dot(v_norm, Av))

    # DIRECT vector residual (NEVER fl(||Av||^2 - theta^2))
    r_var = Av - (theta * v_norm)
    eta = float(np.linalg.norm(r_var))
    eta_sq = eta ** 2

    # Enclosure using module-specific gap
    if gamma_module <= theta:
        enclosure_half_width = float("inf")
        enclosure_valid = False
    else:
        enclosure_half_width = eta_sq / (gamma_module - theta)
        enclosure_valid = True

    solver_rayleigh_discrepancy = abs(lambda_solver - theta)

    return {
        "acceptance_passed": acceptance_passed,
        "scaled_solver_residual": scaled_solver_residual,
        "residual_solver_norm": residual_solver_norm,
        "scale_factor": scale_factor,
        "rayleigh_quotient": theta,
        "direct_residual_eta": eta,
        "enclosure_valid": enclosure_valid,
        "enclosure_half_width": enclosure_half_width,
        "solver_rayleigh_discrepancy": solver_rayleigh_discrepancy
    }
